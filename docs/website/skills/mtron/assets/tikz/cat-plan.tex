% THE FIGURE, IN ONE PICTURE. A path::T's out-edge is a plan::T — a rewrite rule as a
% value, not a closure. The lift alpha is vertical (a function: one abstraction per chain),
% the out-edge is horizontal (the plan is reachable from the path). The shared type method
% lower(plan::T) applies the plan; the plan's selector reads abstraction::T's morphism fields.
%
%       [abstraction::T]   type code — the selector's form/law
%              ^
%         a    |  lift
%       [path::T] --out-edge--> [plan::T]   value code, a rewrite rule as data

\begin{tikzpicture}[

  el/.style={font=\scriptsize, inner sep=1pt},
  sub/.style={font=\tiny, text=white!60!black, align=left, inner sep=1pt},
  color=white
]

% abstraction::T directly above path::T — the squared lift, not a triangle
\node[c] (A) at (0,2.4) {\texttt{abstraction::T}};
\node[sub, anchor=west, align=left] at (1.3,2.4) {type code — the\\ selector's form/law};

% path::T and its out-edge, plan::T
\node[c] (p)  at (0,0) {\texttt{path::T}};
\node[c] (pl) at (4.8,0) {\texttt{plan::T}};

% the lift is vertical (a function: one abstraction per chain)
\draw[e] (p.north) -- node[el,left=2pt]{$\alpha$} (A.south);

% the out-edge is horizontal (the plan is reachable from the path)
\draw[e] (p) -- node[el,above=2pt]{out-edge} (pl);

% what each is
\node[sub, anchor=east, align=right] at (-0.8,0) {value code};
\node[sub, anchor=north, align=center] at (4.8,-0.8) {a rewrite rule as data\\ applied by $\mathrm{lower}(\cdot)$};

\end{tikzpicture}
